/** * Project Name: Junior Cloud * Description: A Cloudflare Worker script to frame and proxy CloudMoon * for smooth cloud gaming in the browser. */ const TARGET_URL = "https://web.cloudmoonapp.com/"; // Official CloudMoon Web App async function handleRequest(request) { const url = new URL(request.url); // 1. If hitting the root or base paths, serve the clean HTML frame if (url.pathname === "/" || url.pathname === "/index.html") { const html = ` Junior Cloud `; return new Response(html, { headers: { "Content-Type": "text/html;charset=UTF-8" }, }); } // 2. Proxy handler to bypass restrictive frame options and load assets safely if (url.pathname.startsWith("/_proxy_/")) { const targetPath = url.pathname.replace("/_proxy_/", ""); const destinationUrl = new URL(targetPath, TARGET_URL); destinationUrl.search = url.search; // Duplicate the request parameters const modifiedRequest = new Request(destinationUrl.toString(), { method: request.method, headers: request.headers, body: request.body, redirect: "follow" }); try { const response = await fetch(modifiedRequest); // Strip out headers that break framing or cross-origin features const newHeaders = new Headers(response.headers); newHeaders.delete("X-Frame-Options"); newHeaders.delete("Content-Security-Policy"); newHeaders.set("Access-Control-Allow-Origin", "*"); return new Response(response.body, { status: response.status, statusText: response.statusText, headers: newHeaders, }); } catch (err) { return new Response("Error connecting to Junior Cloud proxy gateway.", { status: 502 }); } } // 3. Fallback: Directly forward other structural resource assets through the pipeline const assetUrl = new URL(url.pathname, TARGET_URL); assetUrl.search = url.search; return fetch(new Request(assetUrl.toString(), request)); } addEventListener("fetch", event => { event.respondWith(handleRequest(event.request)); });
