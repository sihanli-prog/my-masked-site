// ==UserScript== // @name Junior Cloud Mask // @namespace https://github.com/ // @version 1.1 // @description Forces CrazyGames title and logo to change to Junior Cloud // @author You // @match *://*://* // @run-at document-start // @grant none // ==/UserScript== (function() { 'use strict'; const newTitle = "junior chloud"; const newIconUrl = "https://icons8.com"; // Immediate injection document.title = newTitle; function applyMask() { document.title = newTitle; // Update Favicon Link let link = document.querySelector("link[rel*='icon']"); if (!link) { link = document.createElement('link'); link.rel = 'icon'; document.head.appendChild(link); } link.href = newIconUrl; } // Watch the tab constantly for title changes const observer = new MutationObserver(() => { if (document.title !== newTitle) { document.title = newTitle; } }); observer.observe(document.documentElement, { childList: true, subtree: true }); // Run mask changes as the page elements pop in window.addEventListener('DOMContentLoaded', applyMask); window.addEventListener('load', applyMask); setInterval(applyMask, 500); })();
